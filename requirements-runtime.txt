# Runtime dependencies for the deployed service.
#
# requirements.txt is the development set: it also carries pytest, corner and
# the notebook tooling, none of which the running API imports. This file is
# what the container installs, and it is deliberately the smaller set -- the
# image is already ~400 MB of compiled scientific Python before anything else.
#
# Derived by importing api.main, running one full analysis and loading the
# lazy MAST path, then reading back sys.modules. Transitive dependencies
# (matplotlib, pyarrow, numba, llvmlite, tqdm) are left to the resolver.

# --- core scientific stack ---
numpy>=1.26
scipy>=1.11
pandas>=2.1

# --- astronomy ---
astropy>=6.0             # BoxLeastSquares, LombScargle
wotan>=1.10              # transit-safe detrending (pulls numba + llvmlite)

requests>=2.31           # MAST search and download; see exodet.mast
#
# lightkurve and astroquery are deliberately absent. They do the same job, but
# bring bokeh, matplotlib, PIL, fontTools, botocore and pyvo with them -- about
# 137 MB of image for one button. exodet.mast reaches the same two MAST
# endpoints with requests and reads the FITS with astropy, which is already
# here. scripts/ still use lightkurve, so it stays in requirements.txt.
